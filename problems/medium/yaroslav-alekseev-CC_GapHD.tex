\section{Preliminaries}

For $x,y\in\{0,1\}^n$, let
\[
d_H(x,y):=\bigl|\{i\in[n]:x_i\neq y_i\}\bigr|
\]
denote their Hamming distance.

\begin{definition}[Gap Hamming Distance]
The partial Boolean function
\[
\operatorname{GapHD}_n:\{0,1\}^n\times\{0,1\}^n
    \longrightarrow \{0,1,*\}
\]
is defined by
\[
\operatorname{GapHD}_n(x,y):=
\begin{cases}
0, & d_H(x,y)\le n/3,\\
1, & d_H(x,y)\ge 2n/3,\\
*, & n/3<d_H(x,y)<2n/3.
\end{cases}
\]
\end{definition}

\begin{definition}[Pseudodeterministic communication complexity]
For a total Boolean function
$f:\{0,1\}^n\times\{0,1\}^n\to\{0,1\}$, let
$\mathrm{R}_\varepsilon(f)$ be the minimum worst-case communication cost
of a public-coin randomized protocol that outputs $f(x,y)$ with
probability at least $1-\varepsilon$ on every input $(x,y)$.

A \emph{completion} of a partial Boolean function
$g:\{0,1\}^n\times\{0,1\}^n\to\{0,1,*\}$ is a total Boolean function
$f$ such that $f(x,y)=g(x,y)$ whenever $g(x,y)\neq *$. Define
\[
\mathrm{PsR}_\varepsilon(g)
:=
\min\bigl\{\mathrm{R}_\varepsilon(f):
             f\text{ is a completion of }g\bigr\}.
\]
\end{definition}

\section{Problem formulation}

There exists an absolute constant $c>0$ such that, for all sufficiently
large $n$,
\[
\mathrm{PsR}_{1/3}(\operatorname{GapHD}_n)\ge cn.
\]
Equivalently, every completion
$f:\{0,1\}^n\times\{0,1\}^n\to\{0,1\}$ of
$\operatorname{GapHD}_n$ satisfies
\[
\mathrm{R}_{1/3}(f)\ge cn.
\]
