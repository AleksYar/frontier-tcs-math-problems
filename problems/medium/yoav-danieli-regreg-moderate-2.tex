\section{Atoms and the symbolic alphabet}

\begin{definition}[Atoms and data words]
Let $\mathbb A$ be a countably infinite set of \emph{atoms}. A finite data word
is an element of $\mathbb A^*$.
\end{definition}

For every $i\geq1$, let $x_i$ output the current value of register $i$, and let
$\rho_i$ reassign register $i$. The symbol $x_0$ outputs an atom different from
all currently stored register values. Define
\[
 \Gamma_\infty=\{x_0\}\cup\{x_i,\rho_i:i\geq1\}.
\]
For a finite register set $I\subseteq\mathbb N_{>0}$, let
\[
 \Gamma_I=\{x_0\}\cup\{x_i,\rho_i:i\in I\}.
\]

\section{Scopes and instances}

Let $w=s_1\cdots s_m\in\Gamma_I^*$. For each $i\in I$, the occurrences of
$\rho_i$ divide $w$ into scopes of register $i$: the initial scope is before
the first $\rho_i$, and every later scope starts immediately after a
$\rho_i$ and ends immediately before the next $\rho_i$, or at the end of the
word. Empty scopes are allowed.

\begin{definition}[Legal assignment trace]
A legal assignment trace for $w$ consists of valuations
\[
 \nu_0,\nu_1,\ldots,\nu_m:I\to\mathbb A
\]
and, for every position $p$ with $s_p=x_0$, an atom $b_p\in\mathbb A$, subject
to the following conditions.

\begin{enumerate}[label=(\roman*)]
\item $\nu_0$ is injective.
\item If $s_p=\rho_i$, then
\[
 \nu_p(j)=\nu_{p-1}(j)\quad(j\neq i)
\]
and
\[
 \nu_p(i)\notin\{\nu_{p-1}(j):j\in I\}.
\]
Thus the new value differs from the previous value of register $i$ and from
all current values of the other registers.
\item If $s_p=x_i$, then $\nu_p=\nu_{p-1}$.
\item If $s_p=x_0$, then $\nu_p=\nu_{p-1}$ and
\[
 b_p\notin\nu_{p-1}(I).
\]
Different $x_0$ occurrences are chosen independently. They may equal one
another or values from inactive register scopes, provided they differ from all
register values current at that position.
\end{enumerate}
\end{definition}

\begin{definition}[Instance]
A legal assignment trace produces a data word by scanning $w$ from left to
right and applying
\[
\begin{array}{c|c}
\text{symbol}&\text{output}\\ \hline
x_i&\nu_{p-1}(i)\\
x_0&b_p\\
\rho_i&\varepsilon.
\end{array}
\]
The set of all resulting data words is denoted
$\operatorname{Inst}_I(w)$. For $K\subseteq\Gamma_I^*$, define
\[
 \operatorname{Inst}_I(K)=\bigcup_{w\in K}\operatorname{Inst}_I(w).
\]
\end{definition}

This operational definition is equivalent to assigning one atom to every
scope, with adjacent scopes of the same register assigned different atoms and
all simultaneously current register values pairwise different.

\section{Symbolic automata and symbolic regularity}

\begin{definition}[Symbolic automaton]
For finite $I$, a symbolic automaton is an ordinary finite automaton over
$\Gamma_I$. If it accepts $K\subseteq\Gamma_I^*$, the data language it
generates is $\operatorname{Inst}_I(K)$.
\end{definition}

\begin{definition}[Symbolically regular data language]
A data language $D\subseteq\mathbb A^*$ is \emph{symbolically regular} if
there are finite $I$ and regular $K\subseteq\Gamma_I^*$ such that
\[
 D=\operatorname{Inst}_I(K).
\]
Such a language $K$ is a symbolic representation of $D$.
\end{definition}

\section{Ordinary and generalized symbolic regular expressions}

\begin{definition}[Ordinary expressions]
Ordinary symbolic regular expressions are generated by
\[
 E::=\emptyset\mid\varepsilon\mid s\mid E\cup E\mid E\cdot E\mid E^*,
 \qquad s\in\Gamma_\infty,
\]
with standard semantics $L_{\mathrm s}(E)\subseteq\Gamma_\infty^*$.
\end{definition}

\begin{definition}[Support]
The register support of $E$ is
\[
 \operatorname{supp}(E)=
 \{i\geq1:x_i\text{ or }\rho_i\text{ occurs in }E\}.
\]
The symbol $x_0$ contributes no register index.
\end{definition}

\begin{definition}[Domain-dependent complement]
Generalized symbolic regular expressions additionally allow $\neg E$, with
\[
 L_{\mathrm s}(\neg E)=
 \Gamma_{\operatorname{supp}(E)}^*\setminus L_{\mathrm s}(E).
\]
Thus, if $E$ mentions exactly $x_1,x_3,\rho_1,\rho_3$, then
\[
 L_{\mathrm s}(\neg E)=
 \bigl\{x_0,x_1,x_3,\rho_1,\rho_3\bigr\}^*\setminus L_{\mathrm s}(E).
\]
Complement is local: in $(\neg E)F$, the universe used for $\neg E$ depends
only on $\operatorname{supp}(E)$, not on registers appearing only in $F$.
\end{definition}

\section{Star height}

\begin{definition}[Expression height]
For ordinary expressions,
\[
\begin{aligned}
h(\emptyset)=h(\varepsilon)=h(s)&=0,\\
h(E\cup F)=h(EF)&=\max\{h(E),h(F)\},\\
h(E^*)&=1+h(E).
\end{aligned}
\]
For generalized expressions, additionally $h(\neg E)=h(E)$.
\end{definition}

\begin{definition}[Height of a symbolic language]
For regular $K\subseteq\Gamma_I^*$, define
\[
 h_{\mathrm{sym}}(K)=
 \min\{h(E):E\text{ ordinary and }L_{\mathrm s}(E)=K\},
\]
and
\[
 h_{\mathrm{gsym}}(K)=
 \min\{h(E):E\text{ generalized and }L_{\mathrm s}(E)=K\}.
\]
\end{definition}

\begin{definition}[Data star height]
For a symbolically regular data language $D$, define
\[
 h_{\mathrm{data}}(D)=
 \min\left\{
 h_{\mathrm{sym}}(K):
 \begin{array}{l}
 I\subseteq\mathbb N_{>0}\text{ finite},\\
 K\subseteq\Gamma_I^*\text{ regular},\\
 \operatorname{Inst}_I(K)=D
 \end{array}
 \right\}.
\]
\end{definition}

\begin{definition}[Data generalized star height]
For a symbolically regular data language $D$, define
\[
 h_{\mathrm{gdata}}(D)=
 \min\left\{
 h_{\mathrm{gsym}}(K):
 \begin{array}{l}
 I\subseteq\mathbb N_{>0}\text{ finite},\\
 K\subseteq\Gamma_I^*\text{ regular},\\
 \operatorname{Inst}_I(K)=D
 \end{array}
 \right\}.
\]
\end{definition}

Both minima range over all symbolic representations, including
representations using different register sets.

\section{Research Problems }

\subsection{Easy problems}






\begin{definition}[Generalized star-free symbolic expression]
A generalized symbolic regular expression $F$ is
\emph{generalized star-free} if
\[
  h(F)=0.
\]
\end{definition}







\subsection{Moderate problems}



\begin{conjecture}
The following problem is decidable: 
Input: finite $I$, a DFA $A$ over $\Gamma_I$, and $n\in\mathbb N$.
Decide whether
\[
 h_{\mathrm{gdata}}(\operatorname{Inst}_I(L(A)))\leq n.
\]
\end{conjecture}
